cmake_minimum_required(VERSION 3.24)
project(radshark VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8 /Zc:__cplusplus)
endif()
option(RADSHARK_BUILD_APP "Build the standalone application" ${PROJECT_IS_TOP_LEVEL})
option(RADSHARK_BUILD_TESTS "Build analyzer and UI tests" ${PROJECT_IS_TOP_LEVEL})
option(RADSHARK_BUILD_FWGUI_PLUGIN "Build the FreeWili GUI plugin" OFF)
option(RADSHARK_TEST_ENGINE "Match GUI hosts with ImGui test engine enabled" ON)
set(RADSHARK_WIRESPY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../wirespy" CACHE PATH "Independent wirespy checkout")
set(RADSHARK_IMGUI_TARGET "" CACHE STRING "Existing host ImGui target")
include(cmake/Dependencies.cmake)
include(cmake/Capture.cmake)

add_library(radshark_gui STATIC src/shark_view.cpp src/services.cpp src/portable_ui.cpp)
target_include_directories(radshark_gui PUBLIC src)
target_link_libraries(radshark_gui PUBLIC radshark_core ${RADSHARK_IMGUI_TARGET})

include(cmake/Decoder.cmake)
if(RADSHARK_BUILD_APP)
    include(cmake/Application.cmake)
endif()
if(RADSHARK_BUILD_FWGUI_PLUGIN)
    add_subdirectory(plugins/freewiligui)
endif()
